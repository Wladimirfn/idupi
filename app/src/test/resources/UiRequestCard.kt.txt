/**
 * Inline grace-period card for a pending `ui_request` SSE frame.
 *
 * - The 1s countdown derives from the ABSOLUTE `deadlineAt` timestamp (ms), so
 *   leaving Chat and returning later resumes with the correct remaining time
 *   -- the ViewModel keeps `activeUiRequest` alive and the card is recomposed
 *   with the same `request.deadlineAt`. The server's 120s auto-approve
 *   remains the safety net if the user lets the timer hit zero.
 * - `errorMessage` is shown when the server rejected the last answer (409
 *   stale token / superseded). The dialog stays open so the user can retry;
 *   the ViewModel does NOT clear `activeUiRequest` on 409, only on 200.
 * - INPUT's Cancel button clears the TextField locally; it does NOT cancel
 *   the request. Cancellation = expiry auto-resolution, server-side.
 */
@Composable
fun UiRequestCard(
    request: UiRequest,
    onResponse: (Any) -> Unit,
    errorMessage: String? = null,
) {
    var remainingSeconds by remember(request.id, request.deadlineAt) {
        mutableStateOf(computeRemainingSeconds(request.deadlineAt))
    }
    LaunchedEffect(request.id, request.deadlineAt) {
        while (true) {
            delay(1000L)
            remainingSeconds = computeRemainingSeconds(request.deadlineAt)
        }
    }
    val expired = request.deadlineAt > 0L && remainingSeconds <= 0

    Box(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(12.dp))
            .background(SlateCard)
            .border(1.dp, AccentPurple, RoundedCornerShape(12.dp))
            .padding(16.dp)
    ) {
        Column(verticalArrangement = Arrangement.spacedBy(12.dp)) {
            // Header: dot + title + countdown on the right
            Row(
                modifier = Modifier.fillMaxWidth(),
                verticalAlignment = Alignment.CenterVertically,
                horizontalArrangement = Arrangement.spacedBy(8.dp)
            ) {
                Box(
                    modifier = Modifier
                        .size(8.dp)
                        .clip(RoundedCornerShape(4.dp))
                        .background(AccentPurple)
                )
                Text(
                    text = request.title,
                    color = TextPrimary,
                    fontSize = 15.sp,
                    fontWeight = FontWeight.Bold,
                    modifier = Modifier.weight(1f)
                )
                if (request.deadlineAt > 0L) {
                    Text(
                        text = if (expired) "Expirado" else "${remainingSeconds}s",
                        color = if (expired) StatusError else TextSecondary,
                        fontSize = 12.sp,
                        fontWeight = FontWeight.SemiBold,
                    )
                }
            }

            // Message body
            Text(
                text = request.message,
                color = TextPrimary,
                fontSize = 14.sp,
                lineHeight = 20.sp
            )

            // Actions, dispatched on the method the engine asked for
            when (request.method) {
                UiRequestMethod.CONFIRM -> ConfirmActions(
                    onAccept = { onResponse(true) },
                    onReject = { onResponse(false) },
                    enabled = !expired,
                )
                UiRequestMethod.SELECT -> SelectOptions(
                    options = request.options,
                    onPick = { onResponse(it) },
                    enabled = !expired,
                )
                UiRequestMethod.INPUT -> InputField(
                    onSubmit = { onResponse(it) },
                    enabled = !expired,
                )
            }

            // 409 from the server (stale token / superseded request) surfaces
            // here so the user knows their last answer was rejected. The card
            // itself is still shown because the ViewModel keeps
            // `activeUiRequest` set on 409 -- the spec scenario requires the
            // dialog to stay open for a retry.
            if (!errorMessage.isNullOrBlank()) {
                Text(
                    text = errorMessage,
                    color = StatusError,
                    fontSize = 12.sp,
                    fontWeight = FontWeight.SemiBold,
                )
            }
        }
    }
}

@Composable
private fun ConfirmActions(
    onAccept: () -> Unit,
    onReject: () -> Unit,
    enabled: Boolean,
) {
    Row(
        modifier = Modifier.fillMaxWidth(),
        horizontalArrangement = Arrangement.spacedBy(12.dp, Alignment.End)
    ) {
        OutlinedButton(
            onClick = onReject,
            enabled = enabled,
            colors = ButtonDefaults.outlinedButtonColors(contentColor = TextSecondary),
            shape = RoundedCornerShape(8.dp)
        ) {
            Text("Rechazar", fontWeight = FontWeight.Bold)
        }
        Button(
            onClick = onAccept,
            enabled = enabled,
            colors = ButtonDefaults.buttonColors(
                containerColor = AccentPurple,
                contentColor = TextPrimary
            ),
            shape = RoundedCornerShape(8.dp)
        ) {
            Text("Aceptar", fontWeight = FontWeight.Bold)
        }
    }
}

@Composable
private fun SelectOptions(
    options: List<String>,
    onPick: (String) -> Unit,
    enabled: Boolean,
) {
    Column(
        modifier = Modifier.fillMaxWidth(),
        verticalArrangement = Arrangement.spacedBy(8.dp)
    ) {
        options.forEach { option ->
            Button(
                onClick = { onPick(option) },
                enabled = enabled,
                colors = ButtonDefaults.buttonColors(
                    containerColor = SlateBorder,
                    contentColor = TextPrimary
                ),
                modifier = Modifier.fillMaxWidth(),
                shape = RoundedCornerShape(8.dp)
            ) {
                Text(option, fontWeight = FontWeight.SemiBold)
            }
        }
    }
}

/**
 * Free-text input for an engine `input` request.
 *
 * Submit is disabled when the trimmed text is empty (server validates
 * non-empty); Cancel only clears the local TextField -- the request itself
 * stays alive and the grace-period countdown keeps ticking.
 */
@Composable
private fun InputField(
    onSubmit: (String) -> Unit,
    enabled: Boolean,
) {
    var inputValue by remember { mutableStateOf("") }
    val canSubmit = enabled && inputValue.trim().isNotEmpty()

    Column(
        modifier = Modifier.fillMaxWidth(),
        verticalArrangement = Arrangement.spacedBy(8.dp)
    ) {
        OutlinedTextField(
            value = inputValue,
            onValueChange = { inputValue = it },
            placeholder = {
                Text(
                    "Escribe tu respuesta…",
                    fontSize = 14.sp,
                    color = TextSecondary,
                )
            },
            singleLine = false,
            maxLines = 3,
            enabled = enabled,
            keyboardOptions = KeyboardOptions(imeAction = ImeAction.Send),
            keyboardActions = KeyboardActions(
                onSend = {
                    val trimmed = inputValue.trim()
                    if (trimmed.isNotEmpty()) onSubmit(trimmed)
                }
            ),
            colors = OutlinedTextFieldDefaults.colors(
                focusedBorderColor = AccentPurple,
                unfocusedBorderColor = SlateBorder
            ),
            modifier = Modifier.fillMaxWidth()
        )
        Row(
            modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.spacedBy(12.dp, Alignment.End),
            verticalAlignment = Alignment.CenterVertically,
        ) {
            OutlinedButton(
                onClick = { inputValue = "" },
                enabled = enabled,
                colors = ButtonDefaults.outlinedButtonColors(contentColor = TextSecondary),
                shape = RoundedCornerShape(8.dp)
            ) {
                Icon(
                    Icons.Default.Close,
                    contentDescription = null,
                    modifier = Modifier.size(16.dp)
                )
                Spacer(modifier = Modifier.width(4.dp))
                Text("Cancelar", fontWeight = FontWeight.SemiBold)
            }
            Button(
                onClick = {
                    val trimmed = inputValue.trim()
                    if (trimmed.isNotEmpty()) onSubmit(trimmed)
                },
                enabled = canSubmit,
                colors = ButtonDefaults.buttonColors(
                    containerColor = AccentPurple,
                    contentColor = TextPrimary
                ),
                shape = RoundedCornerShape(8.dp)
            ) {
                Icon(
                    Icons.Default.Send,
                    contentDescription = null,
                    modifier = Modifier.size(16.dp)
                )
                Spacer(modifier = Modifier.width(4.dp))
                Text("Enviar", fontWeight = FontWeight.Bold)
            }
        }
    }
}

private fun computeRemainingSeconds(deadlineAt: Long): Long {
    if (deadlineAt <= 0L) return 0L
    val diffMs = deadlineAt - System.currentTimeMillis()
    return (diffMs / 1000L).coerceAtLeast(0L)
}
